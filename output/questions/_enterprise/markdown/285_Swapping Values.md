## Problem Statement

What is the output of the following code?

#include <stdio.h>
void swap(int&, int&); 
void main() { 
        int a = 10,b=20;
        swap (a++,b++);
        printf("\n%d\t%d\t",a, b);
} 
void swap(int& x, int& y) { 
        x+=2;
        y+=3;
}
